> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working `solve(F)` that takes a polynomial system and returns all solutions, using the CommonSolve.jl interface (`solve`/`init`/`solve!`).

**Architecture:** `solve(prob, alg)` dispatches through CommonSolve's `solve! ∘ init` pattern. A `PolynomialProblem` wraps the polynomials + metadata. `TotalDegree` is the algorithm that builds a straight-line homotopy from a total-degree start system. `init` constructs a `SolveCache` holding the homotopy, tracker, and start solutions. `solve!` iterates over start solutions, tracks each path, and collects results into a `Result` type with accessor functions.

**Tech Stack:** CommonSolve.jl (new dep), existing `system_eval`, `StraightLineHomotopy`, `HomotopyEvaluator`, `Tracker`, `track!`

## PolynomialProblem — wraps a polynomial system for solve().

struct PolynomialProblem{P <: MP.AbstractPolynomialLike}
    polynomials::Vector{P}
    parameters::Vector{Any}      # MP variable type varies
    variables::Vector{Any}       # MP variable type varies
    variable_groups::Union{Nothing, Vector{Vector{Int}}}
end

function PolynomialProblem(
        polys::AbstractVector{<:MP.AbstractPolynomialLike};
        parameters::AbstractVector = _empty_vars(polys),
        variables::AbstractVector = _effective_variables(polys, parameters),
        variable_groups::Union{Nothing, Vector{Vector{Int}}} = nothing,
    )
    return PolynomialProblem(
        collect(polys),
        collect(Any, parameters),
        collect(Any, variables),
        variable_groups,
    )
end
```

Note: `parameters` and `variables` are `Vector{Any}` because the DynamicPolynomials variable type is complex and varies. This is fine since these are only used at construction time, never on hot paths.

## PathResult — immutable result from tracking a single path.

@enumx PathResultCode::Int8 begin
    PATH_SUCCESS
    PATH_AT_INFINITY
    PATH_TERMINATED_ACCURACY
    PATH_TERMINATED_ILL_CONDITIONED
    PATH_TERMINATED_MAX_STEPS
    PATH_TERMINATED_STEP_SIZE
    PATH_TERMINATED_INVALID_START
end

struct PathResult
    return_code::PathResultCode.T
    solution::Vector{ComplexF64}
    t::Float64
    accuracy::Float64
    condition_jacobian::Float64
    winding_number::Int
    accepted_steps::Int
    rejected_steps::Int
end

is_success(r::PathResult)::Bool = r.return_code == PathResultCode.PATH_SUCCESS

function is_real(r::PathResult; tol::Float64 = 1.0e-6)::Bool
    return is_success(r) && all(x -> abs(imag(x)) < tol * max(1.0, abs(x)), r.solution)
end

function _tracker_code_to_path_code(code::TrackerCode.T)::PathResultCode.T
    if code == TrackerCode.TRACKER_SUCCESS
        return PathResultCode.PATH_SUCCESS
    elseif code == TrackerCode.TERMINATED_MAX_STEPS
        return PathResultCode.PATH_TERMINATED_MAX_STEPS
    elseif code == TrackerCode.TERMINATED_ACCURACY_LIMIT
        return PathResultCode.PATH_TERMINATED_ACCURACY
    elseif code == TrackerCode.TERMINATED_ILL_CONDITIONED
        return PathResultCode.PATH_TERMINATED_ILL_CONDITIONED
    elseif code == TrackerCode.TERMINATED_INVALID_STARTVALUE
        return PathResultCode.PATH_TERMINATED_INVALID_START
    elseif code == TrackerCode.TERMINATED_STEP_SIZE_TOO_SMALL
        return PathResultCode.PATH_TERMINATED_STEP_SIZE
    else
        return PathResultCode.PATH_TERMINATED_MAX_STEPS
    end
end

function PathResult(tracker::Tracker)
    state = tracker.state
    return PathResult(
        _tracker_code_to_path_code(state.code),
        Vector{ComplexF64}(state.x),
        real(state.segment.t),
        state.accuracy,
        state.cond_J_ẋ,
        0,  # winding_number (no endgame yet)
        state.accepted_steps,
        state.rejected_steps,
    )
end
```

### Task 3: Total Degree Start System

**Files:**
- Create: `src/solving/total_degree.jl`
- Modify: `src/HomotopyContinuationNext.jl`

The total degree start system for polynomials of degrees d₁,...,dₙ is:
  G(x) = [x₁^d₁ - 1, x₂^d₂ - 1, ..., xₙ^dₙ - 1]

The solutions of G are all products of roots of unity:
  x = (ζ₁^k₁, ζ₂^k₂, ..., ζₙ^kₙ) for kᵢ ∈ 0:dᵢ-1
where ζᵢ = exp(2πi/dᵢ).

Total number of paths = ∏ dᵢ (Bézout bound).

- [ ] **Step 1: Create total_degree.jl**

```julia
## TotalDegree — Bézout bound start system for polynomial solving.

struct TotalDegree end

"""
    total_degree_startsystem(degrees, variables) → (polys, SystemEvaluator)

Build the total degree start system G(x) = [x₁^d₁ - 1, ..., xₙ^dₙ - 1]
and its SystemEvaluator.
"""
function total_degree_startsystem(
        degrees::Vector{Int},
        variables::AbstractVector,
    )
    n = length(degrees)
    @assert length(variables) == n "degrees and variables must have same length"
    polys = [variables[i]^degrees[i] - 1 for i in 1:n]
    _, evaluator = system_eval(polys; variables = variables)
    return polys, evaluator
end

"""
    total_degree_solutions(degrees) → Vector{Vector{ComplexF64}}

Generate all ∏dᵢ start solutions for the total degree start system.
Each solution is (ζ₁^k₁, ..., ζₙ^kₙ) where ζᵢ = exp(2πi/dᵢ).
"""
function total_degree_solutions(degrees::Vector{Int})::Vector{Vector{ComplexF64}}
    n = length(degrees)
    # Roots of unity for each variable
    roots = [cis.(2π .* (0:(d - 1)) ./ d) for d in degrees]
    # Cartesian product
    solutions = Vector{ComplexF64}[]
    for combo in Iterators.product(roots...)
        push!(solutions, ComplexF64[combo...])
    end
    return solutions
end

"""
    total_degree_count(degrees) → Int

The Bézout number = ∏dᵢ.
"""
total_degree_count(degrees::Vector{Int})::Int = prod(degrees)
```

## Result — collection of PathResults from a solve() call.

struct Result
    path_results::Vector{PathResult}
    tracked_paths::Int
    seed::UInt32
end

function Base.show(io::IO, r::Result)
    n_success = count(is_success, r.path_results)
    n_real = count(is_real, r.path_results)
    print(io, "Result with ", r.tracked_paths, " tracked paths\n")
    print(io, " • ", n_success, " solutions (", n_real, " real)\n")
    n_failed = r.tracked_paths - n_success
    if n_failed > 0
        print(io, " • ", n_failed, " paths failed")
    end
    return
end

"""
    solutions(result; only_real=false, real_tol=1e-6) → Vector{Vector{ComplexF64}}

Extract solutions from result. If `only_real=true`, only return real solutions.
"""
function solutions(
        r::Result;
        only_real::Bool = false,
        real_tol::Float64 = 1.0e-6,
    )::Vector{Vector{ComplexF64}}
    filter_fn = if only_real
        pr -> is_success(pr) && is_real(pr; tol = real_tol)
    else
        is_success
    end
    return [pr.solution for pr in r.path_results if filter_fn(pr)]
end

"""
    real_solutions(result; tol=1e-6) → Vector{Vector{Float64}}

Extract real-valued solutions as `Vector{Float64}`.
"""
function real_solutions(r::Result; tol::Float64 = 1.0e-6)::Vector{Vector{Float64}}
    return [Float64.(real.(pr.solution)) for pr in r.path_results
            if is_success(pr) && is_real(pr; tol = tol)]
end

"""
    nsolutions(result) → Int
"""
nsolutions(r::Result)::Int = count(is_success, r.path_results)

"""
    nreal(result; tol=1e-6) → Int
"""
nreal(r::Result; tol::Float64 = 1.0e-6)::Int =
    count(pr -> is_success(pr) && is_real(pr; tol = tol), r.path_results)
```

```julia
## solve — CommonSolve.jl integration for polynomial system solving.

"""
    SolveCache

Holds everything needed to track all paths. Created by `init`, consumed by `solve!`.
"""
struct SolveCache
    tracker::Tracker
    start_solutions::Vector{Vector{ComplexF64}}
    seed::UInt32
end

# ── CommonSolve.init: PolynomialProblem + TotalDegree ──

function CommonSolve.init(
        prob::PolynomialProblem, ::TotalDegree;
        seed::UInt32 = rand(Random.RandomDevice(), UInt32),
        tracker_options::TrackerOptions = TrackerOptions(),
    )::SolveCache
    info, eval_F = system_eval(
        prob.polynomials; parameters = prob.parameters, variables = prob.variables,
    )

    _, eval_G = total_degree_startsystem(info.degrees, prob.variables)
    starts = total_degree_solutions(info.degrees)

    rng = Random.MersenneTwister(seed)
    γ = cis(2π * rand(rng))
    H = StraightLineHomotopy(eval_G, eval_F; γ = γ)
    heval = HomotopyEvaluator(H)
    tracker = Tracker(heval; options = tracker_options)

    return SolveCache(tracker, starts, seed)
end

# ── CommonSolve.solve!: track all paths ──

function CommonSolve.solve!(cache::SolveCache)::Result
    tracker = cache.tracker
    path_results = PathResult[]
    sizehint!(path_results, length(cache.start_solutions))

    for x₀ in cache.start_solutions
        track!(tracker, x₀)
        push!(path_results, PathResult(tracker))
    end

    return Result(path_results, length(cache.start_solutions), cache.seed)
end

# ── Convenience: solve(polys; kwargs...) ──

"""
    solve(polynomials; parameters=[], variables=..., seed=..., tracker_options=...)

Solve a polynomial system using total degree homotopy.

# Examples
```julia
@polyvar x y
result = solve([x^2 + y - 1, x*y - 2])
solutions(result)
real_solutions(result)
```
"""
function CommonSolve.solve(
        polys::AbstractVector{<:MP.AbstractPolynomialLike};
        parameters::AbstractVector = _empty_vars(polys),
        variables::AbstractVector = _effective_variables(polys, parameters),
        seed::UInt32 = rand(Random.RandomDevice(), UInt32),
        tracker_options::TrackerOptions = TrackerOptions(),
    )::Result
    prob = PolynomialProblem(polys; parameters = parameters, variables = variables)
    return CommonSolve.solve!(
        CommonSolve.init(prob, TotalDegree(); seed = seed, tracker_options = tracker_options),
    )
end
```

using Test
using HomotopyContinuationNext
using HomotopyContinuationNext: PolynomialProblem, TotalDegree, Result, PathResult,
    PathResultCode, TrackerOptions, TrackerCode,
    solutions, real_solutions, nsolutions, nreal, is_success, is_real,
    total_degree_solutions, total_degree_count
using DynamicPolynomials: @polyvar
using CommonSolve: CommonSolve

@testset "Solve" begin

    @testset "total_degree_solutions" begin
        # degrees [2, 3] → 6 solutions
        starts = total_degree_solutions([2, 3])
        @test length(starts) == 6
        # Each start solution has roots of unity
        for s in starts
            @test length(s) == 2
            @test abs(s[1]^2 - 1) < 1e-14
            @test abs(s[2]^3 - 1) < 1e-14
        end
        @test total_degree_count([2, 3]) == 6
    end

    @testset "solve: linear system" begin
        @polyvar x y
        result = solve([x - 2, y - 3])
        @test nsolutions(result) == 1
        sols = solutions(result)
        @test length(sols) == 1
        @test abs(sols[1][1] - 2) < 1e-8
        @test abs(sols[1][2] - 3) < 1e-8
    end

    @testset "solve: quadratic system" begin
        @polyvar x y
        result = solve([x^2 + y - 1, x * y - 0.5])
        @test nsolutions(result) >= 2
        for sol in solutions(result)
            @test abs(sol[1]^2 + sol[2] - 1) < 1e-6
            @test abs(sol[1] * sol[2] - 0.5) < 1e-6
        end
    end

    @testset "solve: katsura-3" begin
        @polyvar x0 x1 x2 x3
        F = [
            x0 + 2x1 + 2x2 + 2x3 - 1,
            x0^2 + 2x1^2 + 2x2^2 + 2x3^2 - x0,
            2x0 * x1 + 2x1 * x2 + 2x2 * x3 - x1,
            x1^2 + 2x0 * x2 + 2x1 * x3 - x2,
        ]
        result = solve(F)
        @test nsolutions(result) >= 2
        for sol in solutions(result)
            residual = maximum(abs.(
                [sol[1] + 2sol[2] + 2sol[3] + 2sol[4] - 1,
                 sol[1]^2 + 2sol[2]^2 + 2sol[3]^2 + 2sol[4]^2 - sol[1],
                 2sol[1]*sol[2] + 2sol[2]*sol[3] + 2sol[3]*sol[4] - sol[2],
                 sol[2]^2 + 2sol[1]*sol[3] + 2sol[2]*sol[4] - sol[3]]
            ))
            @test residual < 1e-6
        end
    end

    @testset "solve: real_solutions" begin
        @polyvar x y
        result = solve([x^2 - 1, y^2 - 4])
        rsols = real_solutions(result)
        @test length(rsols) >= 2  # At least (1,2) and (-1,-2) etc.
        for sol in rsols
            @test sol isa Vector{Float64}
            @test abs(sol[1]^2 - 1) < 1e-6
            @test abs(sol[2]^2 - 4) < 1e-6
        end
    end

    @testset "solve: reproducible with seed" begin
        @polyvar x y
        F = [x^2 + y - 1, x * y - 0.5]
        r1 = solve(F; seed = UInt32(42))
        r2 = solve(F; seed = UInt32(42))
        @test nsolutions(r1) == nsolutions(r2)
        s1 = sort(solutions(r1); by = s -> (real(s[1]), imag(s[1])))
        s2 = sort(solutions(r2); by = s -> (real(s[1]), imag(s[1])))
        for (a, b) in zip(s1, s2)
            @test a ≈ b atol = 1e-10
        end
    end

    @testset "solve: CommonSolve interface" begin
        @polyvar x y
        prob = PolynomialProblem([x^2 - 1, y - 2])
        cache = CommonSolve.init(prob, TotalDegree())
        result = CommonSolve.solve!(cache)
        @test nsolutions(result) >= 1
    end

    @testset "Result: show" begin
        @polyvar x y
        result = solve([x - 1, y - 2])
        buf = IOBuffer()
        show(buf, result)
        s = String(take!(buf))
        @test contains(s, "tracked paths")
        @test contains(s, "solutions")
    end
end
